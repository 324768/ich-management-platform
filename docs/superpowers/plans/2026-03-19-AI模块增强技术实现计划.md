## Chunk 1: Prompt Caching（改动最小、省钱最多）

### 1.1 目标
为三个 Brain 的 SystemMessage 添加 Prompt Cache 支持，命中缓存时降低90%输入token费用。

### 1.2 关键修正
**重要陷阱（必须知道）：**
1. LangChain4j 没有 `.cacheable()` 方法，正确API是 `.cacheSystemMessages(true)` 和 `.cacheTools(true)`
2. `.cacheTools(true)` 有已知bug，会为每个工具单独创建cache block，超过4个上限会抛异常
3. 如果 SystemMessage 包含动态内容（{{skills}}/{{userProfile}}），每次都会cache miss，反而付1.25倍费用
4. **正确做法**：把动态变量从SystemMessage移到UserMessage，保持SystemMessage完全静态

```java
package com.hyang.ich.omnitrix.infrastructure.context;

import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

/**
 * 上下文压缩器 - 将长工具返回结果压缩后再注入上下文
 * 避免原样塞入导致上下文过长
 */
@Slf4j
@Component
public class ContextCompressor {

    private static final int MAX_ORIGINAL_LENGTH = 2000;
    private static final int COMPRESSED_LENGTH = 500;

    /**
     * 压缩工具返回结果
     * 如果结果超过阈值，保留关键信息并压缩
     */
    public String compress(String original, String type) {
        if (original == null || original.length() <= MAX_ORIGINAL_LENGTH) {
            return original;
        }

        // 根据类型选择压缩策略
        return switch (type) {
            case "search_result" -> compressSearchResult(original);
            case "product_list" -> compressProductList(original);
            case "user_profile" -> compressUserProfile(original);
            default -> compressGeneric(original);
        };
    }

    private String compressSearchResult(String original) {
        // 保留前3条结果的完整信息，后续压缩
        String[] lines = original.split("\n");
        StringBuilder sb = new StringBuilder();
        int count = 0;
        for (String line : lines) {
            if (line.trim().isEmpty()) continue;
            if (count < 3) {
                sb.append(line).append("\n");
                count++;
            } else if (count < 10) {
                // 后续只保留关键字段
                sb.append("[更多结果...] ").append(line.substring(0, Math.min(50, line.length()))).append("\n");
                count++;
            }
        }
        return sb.toString();
    }

    private String compressProductList(String original) {
        // 商品列表：只保留名称、价格、关键属性
        return original.lines()
                .filter(line -> !line.isEmpty())
                .limit(10)
                .reduce((a, b) -> a + "\n" + b)
                .orElse(original);
    }

    private String compressUserProfile(String original) {
        // 用户画像：保留最近3条
        String[] lines = original.split("\n");
        StringBuilder sb = new StringBuilder();
        int count = 0;
        for (String line : lines) {
            if (line.trim().isEmpty()) continue;
            if (count < 3) {
                sb.append(line).append("\n");
                count++;
            }
        }
        return sb.toString();
    }

    private String compressGeneric(String original) {
        // 通用压缩：保留首尾，中间压缩
        if (original.length() <= MAX_ORIGINAL_LENGTH) return original;
        String head = original.substring(0, COMPRESSED_LENGTH);
        String tail = original.substring(original.length() - COMPRESSED_LENGTH);
        return head + "\n[...内容已压缩...]\n" + tail;
    }
}
```

```java
package com.hyang.ich.omnitrix.infrastructure.context;

import org.springframework.stereotype.Component;

/**
 * 上下文策略 - 根据问题类型选择不同的上下文长度和内容
 */
@Component
public class ContextStrategy {

    /**
     * 问题类型枚举
     */
    public enum QueryType {
        SIMPLE_FACT,        // 简单事实查询（短上下文）
        COMPLEX_ANALYSIS,   // 复杂分析（长上下文）
        SHOPPING,           // 购物相关（中上下文）
        RECOMMENDATION,     // 推荐相关（中上下文）
        GENERAL             // 一般对话（默认上下文）
    }

    /**
     * 判断问题类型（使用contains替代正则，更可靠）
     */
    public QueryType classify(String query) {
        if (query == null || query.isEmpty()) {
            return QueryType.GENERAL;
        }
        
        String lower = query.toLowerCase();
        
        // 简单事实查询
        if (lower.contains("什么是") || lower.contains("哪一年") || 
            lower.contains("谁发明") || lower.contains("何时") || lower.contains("哪里")) {
            return QueryType.SIMPLE_FACT;
        }
        
        // 复杂分析
        if (lower.contains("分析") || lower.contains("比较") || 
            lower.contains("对比") || lower.contains("为什么") || lower.contains("如何实现") || 
            lower.contains("原理")) {
            return QueryType.COMPLEX_ANALYSIS;
        }
        
        // 购物相关
        if (lower.contains("购买") || lower.contains("买") || 
            lower.contains("价格") || lower.contains("多少钱") || lower.contains("优惠") || 
            lower.contains("下单")) {
            return QueryType.SHOPPING;
        }
        
        // 推荐相关
        if (lower.contains("推荐") || lower.contains("相似") || 
            lower.contains("类似") || lower.contains("喜欢") || lower.contains("感兴趣")) {
            return QueryType.RECOMMENDATION;
        }
        
        return QueryType.GENERAL;
    }

    /**
     * 获取该问题类型的上下文预算（token）
     */
    public int getContextBudget(QueryType type) {
        return switch (type) {
            case SIMPLE_FACT -> 2000;
            case COMPLEX_ANALYSIS -> 8000;
            case SHOPPING -> 4000;
            case RECOMMENDATION -> 4000;
            case GENERAL -> 6000;
        };
    }

    /**
     * 是否需要注入用户画像
     */
    public boolean needsUserProfile(QueryType type) {
        return type == QueryType.RECOMMENDATION || type == QueryType.SHOPPING;
    }

    /**
     * 是否需要注入 Skills
     */
    public boolean needsSkills(QueryType type) {
        return type != QueryType.SIMPLE_FACT;
    }
}
```

```java
package com.hyang.ich.omnitrix.infrastructure.rag;

import com.hyang.ich.omnitrix.infrastructure.llm.LlmProperties;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.Duration;

/**
 * 查询改写器 - 当检索结果不相关时，改写查询重新检索
 */
@Slf4j
@Component
public class QueryRewriter {

    private final ChatLanguageModel chatModel;

    public QueryRewriter(LlmProperties properties) {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(properties.getAuxiliaryConfig().getApiKey())
                .modelName(properties.getAuxiliaryConfig().getModel())
                .baseUrl(properties.getAuxiliaryConfig().getApiUrl())
                .temperature(0.3)
                .maxTokens(500)
                .timeout(Duration.ofSeconds(30))
                .build();
    }

    /**
     * 改写查询
     * @param originalQuery 原始查询
     * @param searchResults 当前检索结果（用于分析为什么检索不到）
     * @return 改写后的查询
     */
    public String rewrite(String originalQuery, String searchResults) {
        String prompt = String.format("""
                你是一个查询改写专家。当用户的问题无法从知识库检索到相关内容时，需要改写查询使其更容易找到答案。

                原始查询：%s

                当前检索结果（不相关）：
                %s

                请分析为什么检索不到，并给出3个可能的改进查询词（用逗号分隔）。
                只输出查询词，不要有其他内容。
                """, originalQuery, searchResults);

        try {
            String rewrite = chatModel.chat(prompt);
            // 取第一行作为改写后的查询
            String[] lines = rewrite.split("\n");
            for (String line : lines) {
                line = line.trim();
                if (!line.isEmpty() && !line.startsWith("可能的")) {
                    log.info("查询改写: {} -> {}", originalQuery, line);
                    return line;
                }
            }
        } catch (Exception e) {
            log.warn("查询改写失败: {}", e.getMessage());
        }

        // 失败时添加关键词重试
        return originalQuery + " 非遗";
    }
}
```

```java
package com.hyang.ich.omnitrix.infrastructure.rag;

import com.hyang.ich.omnitrix.infrastructure.llm.LlmProperties;
import com.hyang.ich.omnitrix.infrastructure.vector.QdrantVectorStore;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.util.List;

/**
 * 相关性评估器 - 使用 LLM 判断检索结果是否回答了问题
 */
@Slf4j
@Component
public class RelevanceEvaluator {

    private final ChatLanguageModel chatModel;

    public RelevanceEvaluator(LlmProperties properties) {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(properties.getAuxiliaryConfig().getApiKey())
                .modelName(properties.getAuxiliaryConfig().getModel())
                .baseUrl(properties.getAuxiliaryConfig().getApiUrl())
                .temperature(0.1)
                .maxTokens(200)
                .timeout(Duration.ofSeconds(30))
                .build();
    }

    /**
     * 评估检索结果是否与问题相关
     * @return true 表示相关，false 表示不相关
     */
    public boolean evaluate(String query, List<QdrantVectorStore.SearchResult> results) {
        if (results == null || results.isEmpty()) {
            return false;
        }

        // 构建上下文
        StringBuilder context = new StringBuilder();
        for (int i = 0; i < Math.min(3, results.size()); i++) {
            context.append("【内容").append(i + 1).append("】")
                   .append(results.get(i).getText())
                   .append("\n");
        }

        String prompt = String.format("""
                你是一个相关性评估专家。判断给定的知识库内容是否能够回答用户的问题。

                用户问题：%s

                知识库内容：
                %s

                请判断：这些知识库内容是否能回答用户的问题？
                只输出"是"或"否"，不要有其他内容。
                """, query, context.toString());

        try {
            String answer = chatModel.chat(prompt).trim();
            boolean relevant = answer.startsWith("是") || answer.equalsIgnoreCase("yes");
            log.debug("相关性评估: query={}, relevant={}", query, relevant);
            return relevant;
        } catch (Exception e) {
            log.warn("相关性评估失败: {}", e.getMessage());
            // 失败时保守返回true，让后续生成处理
            return true;
        }
    }
}
```

```java
package com.hyang.ich.omnitrix.infrastructure.rag;

import com.hyang.ich.omnitrix.infrastructure.llm.LlmProperties;
import dev.langchain4j.model.chat.ChatLanguageModel;
import dev.langchain4j.model.openai.OpenAiChatModel;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.util.List;
import java.util.stream.Collectors;

/**
 * 幻觉检查器 - 检查 LLM 生成的回答是否基于知识库内容
 */
@Slf4j
@Component
public class HallucinationChecker {

    private final ChatLanguageModel chatModel;

    public HallucinationChecker(LlmProperties properties) {
        this.chatModel = OpenAiChatModel.builder()
                .apiKey(properties.getAuxiliaryConfig().getApiKey())
                .modelName(properties.getAuxiliaryConfig().getModel())
                .baseUrl(properties.getAuxiliaryConfig().getApiUrl())
                .temperature(0.1)
                .maxTokens(300)
                .timeout(Duration.ofSeconds(30))
                .build();
    }

    /**
     * 检查回答是否基于知识库内容
     * @return 检查结果，包含是否通过和修改建议
     */
    public CheckResult check(String answer, String query, List<String> sources) {
        String sourceTexts = sources.stream()
                .limit(3)
                .collect(Collectors.joining("\n"));

        String prompt = String.format("""
                你是一个幻觉检测专家。判断AI生成的回答是否基于提供的知识库内容，而不是凭空编造。

                用户问题：%s

                知识库内容：
                %s

                AI生成的回答：
                %s

                请判断：
                1. 回答是否基于知识库内容？（是/否）
                2. 如果不是，修改建议是什么？

                格式：
                是否基于知识库：是/否
                修改建议：[如果有问题，给出修改建议]
                """, query, sourceTexts, answer);

        try {
            String result = chatModel.chat(prompt);
            boolean passed = result.contains("是") && !result.contains("否");
            
            String suggestion = "";
            if (!passed) {
                // 提取修改建议
                String[] lines = result.split("\n");

                        suggestion = line.replace("修改建议：", "").trim();
                        break;
                    }
                }
            }
            
            return new CheckResult(passed, suggestion);
        } catch (Exception e) {
            log.warn("幻觉检查失败: {}", e.getMessage());
            // 失败时保守返回通过
            return new CheckResult(true, "");
```

```java
/**

public RagResult answerAgentic(String query, Integer topK, Double scoreThreshold) {
    if (query == null || query.trim().isEmpty()) {
        return RagResult.builder()
                .answer("请提供有效的问题")
                .sources(Collections.emptyList())
                .build();
    }

    String currentQuery = query;
    List<QdrantVectorStore.SearchResult> finalResults = null;
    String finalAnswer = null;
    List<String> sourceTexts;

    // 循环：检索 -> 评估 -> 如果不通过则改写重试
    for (int attempt = 0; attempt < MAX_RETRIES; attempt++) {
        log.debug("Agentic RAG 第{}次尝试: query={}", attempt + 1, currentQuery);

        // 1. 向量检索
        List<QdrantVectorStore.SearchResult> searchResults = 
                vectorStore.search(currentQuery, topK, scoreThreshold);

        if (searchResults.isEmpty()) {
            if (attempt == 0) {
                // 第一次检索为空，尝试扩展查询
                currentQuery = query + " 非遗";
                continue;
            }
            break;
        }

        // 2. LLM 评估检索结果是否相关
        boolean isRelevant = relevanceEvaluator.evaluate(currentQuery, searchResults);
        
        if (!isRelevant) {
            // 不相关，改写查询再检索
            String searchResultText = searchResults.stream()
                    .map(QdrantVectorStore.SearchResult::getText)
                    .collect(Collectors.joining("\n"));
            currentQuery = queryRewriter.rewrite(query, searchResultText);
            log.info("检索不相关，改写查询: {} -> {}", currentQuery, currentQuery);
            continue;
        }

        // 3. 相关，构建上下文并生成回答
        String context = buildContext(searchResults);
        String prompt = buildPrompt(query, context);
        
        try {
            String answer = chatModel.chat(prompt);
            
            // 4. 幻觉检查
            sourceTexts = searchResults.stream()
                    .map(QdrantVectorStore.SearchResult::getText)
                    .collect(Collectors.toList());
            
            HallucinationChecker.CheckResult checkResult = 
                    hallucinationChecker.check(answer, query, sourceTexts);
            
            if (checkResult.passed()) {
                // 通过检查，返回结果
                finalResults = searchResults;
                finalAnswer = answer;
                break;
            } else {
                // 幻觉检查不通过，尝试修改回答
                if (checkResult.suggestion() != null && !checkResult.suggestion().isEmpty()) {
                    finalAnswer = checkResult.suggestion();
                    finalResults = searchResults;
                    break;
                }
            }
        } catch (Exception e) {
            log.error("RAG生成失败: {}", e.getMessage());
        }
    }

    // 返回最终结果
    if (finalAnswer == null) {
        // 所有尝试都失败，尝试直接生成
        if (finalResults == null || finalResults.isEmpty()) {
            return RagResult.builder()
                    .answer("抱歉，知识库中没有找到与您问题相关的内容。")
                    .sources(Collections.emptyList())
                    .build();
        }
        String context = buildContext(finalResults);
        String prompt = buildPrompt(query, context);
        try {
            finalAnswer = chatModel.chat(prompt);
        } catch (Exception e) {
            finalAnswer = "抱歉，处理您的问题时出现错误。";
        }
    }

    // 构建来源
    List<Source> sources = finalResults.stream()
            .map(r -> Source.builder()
                    .id(r.getId())
                    .text(r.getText())
                    .score(r.getScore())
                    .metadata(r.getMetadata())
                    .build())
            .collect(Collectors.toList());

    return RagResult.builder()
            .answer(finalAnswer)
            .sources(sources)
            .build();
}
```

```java
/**
 * Graphiti 时序知识图谱服务
 * 每条事实都有一个"有效时间窗口"，支持时序演化
 */
@Slf4j
@Component
@ConditionalOnProperty(prefix = "omnitrix.graphiti", name = "enabled", havingValue = "true")
public class GraphitiService {

    private final Driver driver;
    private final GraphitiProperties properties;

    public GraphitiService(GraphitiProperties properties) {
        this.properties = properties;
        this.driver = GraphDatabase.driver(
                properties.getUri(),
                Auth.basic(properties.getUsername(), properties.getPassword())
        );
    }

    /**
     * 关闭驱动连接
     */
    @PreDestroy
    public void close() {
        if (driver != null) {
            try {
                driver.close();
                log.info("Neo4j 驱动已关闭");
            } catch (Exception e) {
                log.warn("关闭 Neo4j 驱动失败: {}", e.getMessage());
            }
        }
    }

    /**
     * 添加用户偏好事实（带时间戳）
     */
    public void addUserPreference(Long userId, String entity, String relation, String value) {
        String cypher = """
            MERGE (u:User {id: $userId})
            MERGE (e:Entity {name: $entity, type: $relation})
            SET e.value = $value, e.updatedAt = datetime()
            MERGE (u)-[r:PREFERS]->(e)
            SET r.value = $value, r.updatedAt = datetime()
            """;

        try (Session session = driver.session()) {
            session.run(cypher,
                    Values.parameters("userId", userId.toString(),
                            "entity", entity,
                            "relation", relation,
                            "value", value));
            log.debug("添加用户偏好: userId={}, entity={}, relation={}, value={}",
                    userId, entity, relation, value);
        } catch (Exception e) {
            log.error("添加用户偏好失败: {}", e.getMessage());
        }
    }

    /**
     * 获取用户当前偏好（最新）
     */
    public Map<String, String> getCurrentPreferences(Long userId) {
        String cypher = """
            MATCH (u:User {id: $userId})-[r:PREFERS]->(e:Entity)
            RETURN e.name as entity, e.value as value, e.updatedAt as updatedAt
            ORDER BY e.updatedAt DESC
            """;

        Map<String, String> prefs = new HashMap<>();
        try (Session session = driver.session()) {
            Result result = session.run(cypher,
                    Values.parameters("userId", userId.toString()));
            while (result.hasNext()) {
                Record record = result.next();
                String entity = record.get("entity").asString();
                String value = record.get("value").asString();
                prefs.put(entity, value);
            }
        } catch (Exception e) {
            log.error("获取用户偏好失败: {}", e.getMessage());
        }
        return prefs;
    }

    /**
     * 获取用户偏好历史（时序）
     */
    public List<PreferenceHistory> getPreferenceHistory(Long userId, String entityType) {
        String cypher = """
            MATCH (u:User {id: $userId})-[r:PREFERS]->(e:Entity {type: $entityType})
            RETURN e.name as entity, r.value as value, r.updatedAt as updatedAt
            ORDER BY r.updatedAt DESC
            """;

        List<PreferenceHistory> history = new ArrayList<>();
        try (Session session = driver.session()) {
            Result result = session.run(cypher,
                    Values.parameters("userId", userId.toString(), "entityType", entityType));
            while (result.hasNext()) {
                Record record = result.next();
                history.add(new PreferenceHistory(
                        record.get("entity").asString(),
                        record.get("value").asString(),
                        record.get("updatedAt").asLocalDateTime()
                ));
            }
        } catch (Exception e) {
            log.error("获取偏好历史失败: {}", e.getMessage());
        }
        return history;
    }

    /**
     * 清理过期事实
     */
    public void cleanExpiredFacts() {
        String cypher = """
            MATCH (e:Entity)
            WHERE e.updatedAt < datetime() - duration({days: $ttlDays})
            DETACH DELETE e
            """;

        try (Session session = driver.session()) {
            session.run(cypher,
                    Values.parameters("ttlDays", properties.getFactTtlDays()));
            log.info("清理过期事实完成");
        } catch (Exception e) {
            log.error("清理过期事实失败: {}", e.getMessage());
        }
    }

    public record PreferenceHistory(String entity, String value, LocalDateTime updatedAt) {}
}
```

```java
import dev.langchain4j.mcp.client.DefaultMcpClient;
import dev.langchain4j.mcp.client.McpClient;
import dev.langchain4j.mcp.client.transport.McpTransport;
import dev.langchain4j.mcp.client.transport.stdio.StdioMcpTransport;        // ← 注意 .stdio. 层
import dev.langchain4j.mcp.client.transport.http.StreamableHttpMcpTransport; // ← 正确
import dev.langchain4j.mcp.client.transport.http.HttpMcpTransport;           // ← 旧版SSE兜底
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.stereotype.Component;

import jakarta.annotation.PostConstruct;
import java.util.ArrayList;
import java.util.List;

/**
 * MCP Client 工厂 - 根据配置创建 MCP 客户端
 * 支持 Streamable HTTP 和 STDIO 两种传输模式
 */
@Slf4j
@Component
@ConditionalOnProperty(prefix = "omnitrix.mcp", name = "enabled", havingValue = "true")
public class McpClientFactory {

    private final McpProperties properties;
    private final List<McpClient> mcpClients = new ArrayList<>();

    public McpClientFactory(McpProperties properties) {
        this.properties = properties;
    }

    @PostConstruct
    public void init() {
        if (!properties.isEnabled() || properties.getServers() == null) {
            return;
        }

        for (McpProperties.ServerConfig server : properties.getServers()) {
            try {
                McpClient client = createClient(server);
                if (client != null) {
                    mcpClients.add(client);
                    log.info("MCP Server [{}] 已连接, type={}", server.getName(), server.getType());
                }
            } catch (Exception e) {
                log.error("连接 MCP Server [{}] 失败: {}", server.getName(), e.getMessage());
            }
        }
    }

    /**
     * 创建 MCP Client
     * 注意：部分旧版 MCP Server 不支持 Streamable HTTP，需要兜底重试
     */
    private McpClient createClient(McpProperties.ServerConfig server) {
        // Streamable HTTP 模式（2025年后推荐）
        if ("streamable-http".equals(server.getType())) {
            try {
                return createStreamableHttpClient(server);
            } catch (Exception e) {
                log.warn("Streamable HTTP 连接失败，尝试旧版 SSE 模式: {}", e.getMessage());
                // 兜底：尝试旧版 SSE 传输
                return createSseClient(server);
            }
        }

        // STDIO 模式
        if ("stdio".equals(server.getType())) {
            return createStdioClient(server);
        }

        // 兼容旧版 sse 配置
        if ("sse".equals(server.getType())) {
            return createSseClient(server);
        }

        throw new IllegalArgumentException("不支持的 MCP Server 类型: " + server.getType());
    }

    /**
     * 创建 Streamable HTTP 客户端
     */
    private McpClient createStreamableHttpClient(McpProperties.ServerConfig server) {
        McpTransport transport = new StreamableHttpMcpTransport.Builder()
                .url(server.getUrl())
                .logRequests(server.isLogRequests())
                .logResponses(server.isLogResponses())
                .build();

        return new DefaultMcpClient.Builder()
                .key(server.getName())
                .transport(transport)
                .build();
    }

    /**
     * 创建 STDIO 客户端
     */
    private McpClient createStdioClient(McpProperties.ServerConfig server) {
        List<String> command = buildCommand(server);

        McpTransport transport = new StdioMcpTransport.Builder()
                .command(command)
                .logEvents(server.isLogRequests())
                .build();

        return new DefaultMcpClient.Builder()
                .key(server.getName())
                .transport(transport)
                .build();
    }

    /**
     * 创建 SSE 客户端（旧版兼容）
     */
    @SuppressWarnings("deprecation")
    private McpClient createSseClient(McpProperties.ServerConfig server) {
        // 使用旧版 SSE 传输作为兜底
        McpTransport transport = new HttpMcpTransport.Builder()
                .url(server.getUrl())
                .logRequests(server.isLogRequests())
                .logResponses(server.isLogResponses())
                .build();

        return new DefaultMcpClient.Builder()
                .key(server.getName())
                .transport(transport)
                .build();
    }

    /**
     * 构建命令列表
     */
    private List<String> buildCommand(McpProperties.ServerConfig server) {
        List<String> cmd = new ArrayList<>();
        cmd.add(server.getCommand());
        if (server.getArgs() != null) {
            cmd.addAll(server.getArgs());
        }
        return cmd;
    }

    public List<McpClient> getMcpClients() {
        return mcpClients;
    }
}
```

```java
import dev.langchain4j.mcp.McpToolProvider;  // ← 是 dev.langchain4j.mcp，不是 .mcp.client
import dev.langchain4j.mcp.client.McpClient;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.stereotype.Component;

import jakarta.annotation.PostConstruct;
import java.util.List;

/**
 * MCP ToolProvider 工厂 - 将 MCP 工具转换为 LangChain4j ToolProvider
 */
@Slf4j
@Component
@ConditionalOnProperty(prefix = "omnitrix.mcp", name = "enabled", havingValue = "true")
public class McpToolProviderFactory {

    private final McpClientFactory mcpClientFactory;
    private McpToolProvider toolProvider;  // ← 修正：dev.langchain4j.mcp.client

    public McpToolProviderFactory(McpClientFactory mcpClientFactory) {
        this.mcpClientFactory = mcpClientFactory;
    }

    @PostConstruct
    public void init() {
        List<McpClient> clients = mcpClientFactory.getMcpClients();
        if (clients == null || clients.isEmpty()) {
            log.info("未配置 MCP Servers");
            return;
        }

        // 正确写法：McpToolProvider.builder() 而不是 new ToolProviderBuilder()
        this.toolProvider = McpToolProvider.builder()
                .mcpClients(clients)
                .failIfOneServerFails(false)  // 某个Server失败不影响其他的
                .build();

        log.info("MCP ToolProvider 已创建，包含 {} 个 MCP Clients", clients.size());
    }

    public McpToolProvider getToolProvider() {
        return toolProvider;
    }
}
```

```java
@Autowired(required = false)  // required=false，MCP未启用时不报错
private McpToolProviderFactory mcpToolProviderFactory;

// 在 buildUserBrainStreaming 方法中添加 MCP 工具
public UserMasterBrain buildUserBrainStreaming(Long userId, String sessionId, String modelCode) {
    ChatMemory memory = buildMemory(sessionId);
    StreamingChatLanguageModel streamingModel = getStreamingModel(modelCode);

    var builder = AiServices.builder(UserMasterBrain.class)
            .streamingChatLanguageModel(streamingModel)
            .chatMemory(memory);

    McpToolProvider mcpProvider = mcpToolProviderFactory != null
            ? mcpToolProviderFactory.getToolProvider() : null;

    if (mcpProvider != null) {
        // 有MCP时：只用toolProvider（内置工具暂时无法同时用，等升级1.x）
        builder.toolProvider(mcpProvider);
    } else {
        // 无MCP时：正常用内置tools
        builder.tools(toolRegistry.getToolsForRole("user"));
    }

    return builder.build();
}
```
